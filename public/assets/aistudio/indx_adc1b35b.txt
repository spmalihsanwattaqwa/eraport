function downloadStudentTemplate(){
  const headers = [["NIS", "NISN", "NAMA", "NAMA_AYAH", "JK", "NAMA_KELAS", "TEMPAT_LAHIR", "TANGGAL_LAHIR", "AGAMA", "TELEPON", "ALAMAT", "NAMA_ORTU_WALI"]];
  const example = [["1001", "0012345678", "أحمد زكي", "عبد الله", "L", state.classes[0]?.CLASS_NAME||"1A", "Jakarta", "2015-05-20", "Islam", "08123456789", "Jl. Merdeka No 1", "عبد الله"]];
  const ws = XLSX.utils.aoa_to_sheet(headers.concat(example));
  const wb = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb, ws, "Template_Siswa");
  XLSX.writeFile(wb, "Template_Import_Siswa.xlsx");
}

function openImportStudent(){
  showModal('Import Data Siswa dari Excel',`
    <div class="form-group">
      <label>Pilih File Excel (.xlsx / .csv)</label>
      <input type="file" id="excelFile" class="form-control" accept=".xlsx, .xls, .csv">
      <div class="helper" style="margin-top:8px;">Pastikan terdapat kolom <b>NAMA</b> dan <b>NAMA_AYAH</b> (disarankan menggunakan tulisan Arab agar tampil seragam di rapor Arab).</div>
    </div>
  `, async()=>{
    const fileInput = document.getElementById('excelFile');
    if(!fileInput.files.length){ toast('Silakan pilih file Excel.','err'); return; }
    const btn = document.getElementById('modalSave');
    setBusy(btn, true, 'Meng-import...');
    const reader = new FileReader();
    reader.onload = async (e) => {
      try {
        const data = new Uint8Array(e.target.result);
        const workbook = XLSX.read(data, {type: 'array'});
        const sheetName = workbook.SheetNames[0];
        const rows = XLSX.utils.sheet_to_json(workbook.Sheets[sheetName]);
        if(!rows.length) throw new Error("File kosong atau format salah.");
        
        const res = await gas('importStudentsFromExcel', state.token, rows);
        closeModal();
        toast(`${res.count} data siswa berhasil di-import.`);
        await refreshBootstrap();
        await renderStudents();
      } catch(err) {
        toast(err.message, 'err');
        setBusy(btn, false);
      }
    };
    reader.readAsArrayBuffer(fileInput.files[0]);
  });
}

// Kalkulasi live pratinjau Nilai Akhir per baris saat diketik atau di-paste
function recalcRowFinal(el) {
  const tr = el.closest('tr');
  if(!tr) return;
  const ptsRaw = tr.querySelector('.pts').value.trim();
  const pasRaw = tr.querySelector('.pas').value.trim();
  const display = tr.querySelector('.final-display');
  if(ptsRaw === '' && pasRaw === '') {
    display.textContent = '-';
    return;
  }
  const pasVal = Number(pasRaw) || 0;
  let finalVal = 0;
  if(ptsRaw === '') {
    finalVal = pasVal;
  } else {
    const ptsVal = Number(ptsRaw) || 0;
    finalVal = Math.round(((ptsVal * 0.30) + (pasVal * 0.70)) * 100) / 100;
  }
  display.textContent = finalVal;
}

// Fungsi penanganan Copy-Paste blok per kolom ala Excel
function handleColumnPaste(e, startIdx, fieldType) {
  e.preventDefault();
  const clipboardData = e.clipboardData || window.clipboardData;
  const pastedData = clipboardData.getData('Text');
  if(!pastedData) return;
  
  const lines = pastedData.split(/\r\n|\n|\r/).filter(l => l.trim() !== '');
  const rows = document.querySelectorAll('#scoreTable tbody tr[data-student]');
  let count = 0;
  
  lines.forEach((val, i) => {
    const targetRowIdx = startIdx + i;
    if(rows[targetRowIdx]) {
      const input = rows[targetRowIdx].querySelector('.' + fieldType);
      if(input) {
        const cleanVal = val.replace(',', '.').trim();
        if(!isNaN(cleanVal) && cleanVal !== '') {
          input.value = cleanVal;
          recalcRowFinal(input);
          count++;
        }
      }
    }
  });
  if(count > 0) toast(`${count} nilai berhasil di-paste ke kolom ${fieldType.toUpperCase()}.`);
}
